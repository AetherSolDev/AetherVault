┌────────────────────────────────────────────────────────────────────────────────────────────────────┐
│                                        Important Notes                                            │
├────────────────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                                    │
│  1. Gemini context surcharge: Models using >200K tokens incur higher rates (2x)                   │
│                                                                                                    │
│  2. DeepSeek peak pricing: Applies daily; schedule heavy workloads during off-peak hours           │
│     (outside 9:00-12:00 & 14:00-18:00 Beijing time)                                              │
│                                                                                                    │
│  3. Cache pricing: DeepSeek cache hit is ~2% of base rate; Gemini is ~10% of input                │
│                                                                                                    │
│  4. All prices are in USD per 1 million tokens                                                    │
│                                                                                                    │
│  5. Regional pricing may vary; consult your API provider                                          │
│                                                                                                    │
│  6. Peak hedge: During DeepSeek peak windows (9-12 & 14-18 Beijing), Gemini 2.5 Flash-Lite        │
│     ($0.10/$0.40) is cheaper than DeepSeek V4 Flash peak ($0.28/$0.56) for comparable output.     │
│     Default: DeepSeek off-peak; switch to Gemini Flash-Lite during peak to keep costs flat.      │
│                                                                                                    │
└────────────────────────────────────────────────────────────────────────────────────────────────────┘

┌────────────────────────────────────────────────────────────────────────────────────────────────────┐
│                                      Python Cost Estimator                                        │
├────────────────────────────────────────────────────────────────────────────────────────────────────┤
│                                                                                                    │
│   def estimate_cost(model, input_tokens, output_tokens, is_peak=False):                           │
│       RATES = {                                                                                   │
│           "deepseek-flash": {"input": 0.14, "output": 0.28},                                      │
│           "deepseek-flash-pro": {"input": 0.435, "output": 0.87},                                 │
│           "gemini-31-pro": {"input": 2.00, "output": 12.00},                                      │
│           "gemini-35-flash": {"input": 1.50, "output": 9.00},                                     │
│           "gemini-31-flash-lite": {"input": 0.25, "output": 1.50},                                │
│           "gemini-25-flash": {"input": 0.30, "output": 2.50},                                     │
│           "gemini-25-flash-lite": {"input": 0.10, "output": 0.40},                                │
│       }                                                                                           │
│       input_rate = RATES[model]["input"] / 1_000_000                                              │
│       output_rate = RATES[model]["output"] / 1_000_000                                            │
│       if "deepseek" in model and is_peak:                                                         │
│           input_rate *= 2.0                                                                       │
│           output_rate *= 2.0                                                                      │
│       return (input_tokens * input_rate) + (output_tokens * output_rate)                          │
│                                                                                                    │
